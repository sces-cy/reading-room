const SHEET_ID = 'YOUR_GOOGLE_SHEET_ID'; const SHEET = '排行榜'; // 欄位：A時間戳記 B姓名 C班級 D秒數 E提示次數 F勳章等級 G勳章 H挑戰次數 function doPost(e) { const lock = LockService.getScriptLock(); lock.tryLock(5000); try { const d = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.openById(SHEET_ID); let sh = ss.getSheetByName(SHEET); if (!sh) { sh = ss.insertSheet(SHEET); sh.appendRow(['時間戳記','姓名','班級', '完成時間(秒)','提示次數','勳章等級','勳章','挑戰次數']); sh.getRange(1,1,1,8).setFontWeight('bold').setBackground('#f3e8c0'); } const key = d.name + '||' + d.cls; const allRows = sh.getDataRange().getValues(); let existRow = -1; for (let i = 1; i < allRows.length; i++) { if (allRows[i][1] + '||' + allRows[i][2] === key) { existRow = i + 1; break; } } const now = new Date().toLocaleString('zh-TW'); const isBetter = (nM,nS,nH,oM,oS,oH) => { if (nM !== oM) return nM < oM; if (nS !== oS) return nS < oS; return nH < oH; }; if (existRow === -1) { sh.appendRow([now,d.name,d.cls,d.seconds, d.hints,d.medalLevel,d.medal,1]); return out({success:true,improved:true,isNew:true}); } else { const old = allRows[existRow - 1]; const playCount = (Number(old[7])||1) + 1; const improved = isBetter( d.medalLevel,d.seconds,d.hints, Number(old[5]),Number(old[3]),Number(old[4]) ); if (improved) { sh.getRange(existRow,1,1,8).setValues([[ now,d.name,d.cls,d.seconds, d.hints,d.medalLevel,d.medal,playCount ]]); } else { sh.getRange(existRow,8).setValue(playCount); } return out({success:true,improved,isNew:false,playCount}); } } catch(err) { return out({success:false,error:err.message}); } finally { lock.releaseLock(); } } function doGet(e) { if (e.parameter.action !== 'lb') return out([]); const ss = SpreadsheetApp.openById(SHEET_ID); const sh = ss.getSheetByName(SHEET); if (!sh) return out([]); const rows = sh.getDataRange().getValues().slice(1) .filter(r => r[0]) .map(r => ({ name:r[1],cls:r[2],seconds:+r[3], hints:+r[4],medalLevel:+r[5],medal:r[6], playCount:+r[7]||1 })); return out(rows); } function out(data){ return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
